cmake_minimum_required(VERSION 3.16)
project(clmlx LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
if(NOT APPLE)
  message(FATAL_ERROR "cl-mlx requires macOS with Apple Silicon")
endif()
add_library(clmlx SHARED shim.cpp)
target_include_directories(clmlx PRIVATE "${MLX_C_SOURCE}")
target_link_libraries(clmlx PRIVATE "${MLX_C_BUILD}/libmlxc.dylib")
set_target_properties(clmlx PROPERTIES BUILD_RPATH "${MLX_C_BUILD};${MLX_C_BUILD}/_deps/mlx-build")
